4. 逐个直接等待Action服务器，只查询服务端就绪，不发送目标。复验时运行下列只读客户端片段，stdout保存JSON；任一不就绪立即停止：

```python
import json
import rclpy
from rclpy.action import ActionClient
from nav2_msgs.action import (
    ComputePathThroughPoses, ComputePathToPose, FollowPath,
    NavigateToPose, NavigateThroughPoses,
)

rclpy.init()
node = rclpy.create_node('u01_action_readiness_probe')
results = []
try:
    for name, action_type in [
        ('/compute_path_through_poses', ComputePathThroughPoses),
        ('/compute_path_to_pose', ComputePathToPose),
        ('/follow_path', FollowPath),
        ('/navigate_to_pose', NavigateToPose),
        ('/navigate_through_poses', NavigateThroughPoses),
    ]:
        client = ActionClient(node, action_type, name)
        try:
            ready = client.wait_for_server(timeout_sec=20.0)
            results.append({'action': name, 'ready': ready})
        finally:
            client.destroy()
        if not ready:
            break
finally:
    node.destroy_node()
    rclpy.shutdown()
print(json.dumps(results))
raise SystemExit(0 if len(results) == 5 and all(r['ready'] for r in results) else 1)
```
